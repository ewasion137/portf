// EquinoxOS Monolithic Kernel Entry Sequence
void kmain(void) {
    serial_init(COM1);
    tty_init();                 
    term_print("\n=== EquinoxOS Kernel Starting ===\n");

    /* Validate Limine Higher-Half Direct Map */
    if (hhdm_request.response == NULL) {
        term_print("ERROR: Limine HHDM not available!\n");
        while (1) __asm__("cli; hlt");
    }
    hhdm_offset = hhdm_request.response->offset;

    init_gdt();
    init_sse();
    pmm_init();
    vmm_init();
    init_heap((uint64_t)pmm_alloc_continuous(16384) + hhdm_offset, 64 * 1024 * 1024);

    /* Write-Combining Accelerated VESA Framebuffer */
    struct limine_framebuffer *fb = framebuffer_request.response->framebuffers[0];
    init_vesa((uintptr_t)fb->address, fb->width, fb->height, fb->pitch);
    vmm_remap_fb_wc();

    /* Mount VFS and Dispatch Ring 3 Userspace Environment */
    vfs_init();
    ext2_init();
    fat32_init();
    exec_from_disk("bin/sysgui.elf"); 
    term_print("enGUI spawned as Ring 3 init process\n");
}
// EquantOS Limine Bootstrap & Userland Spawn
void _start(void) {
    serial_init(COM1);
    enable_fpu_sse();
    init_gdt();
    init_idt();

    if (hhdm_request.response == NULL || hhdm_request.response->offset == 0) {
        PANIC("Limine HHDM response is NULL!");
    }
    hhdm_offset = hhdm_request.response->offset;

    if (framebuffer_request.response != NULL && framebuffer_request.response->framebuffer_count > 0) {
        kernel_fb = framebuffer_request.response->framebuffers[0];
        tty_init(kernel_fb->address, kernel_fb->width, kernel_fb->height, kernel_fb->pitch);
    }

    pmm_init();
    vmm_init();
    init_heap(0, 0);
    apic_init();

    asm volatile ("sti");
    serial_puts(COM1, "[KERNEL] Executing Initcalls...\n");
    do_initcalls();

    tty_print("\nWelcome to EquantOS!\n\n");
    kernel_start_userland();

    for (;;) {
        asm volatile ("hlt");
    }
}
